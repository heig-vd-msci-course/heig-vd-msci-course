const cases = document.querySelectorAll(".case");
const message = document.querySelector("#message");
const choixPersonnage = document.querySelector("#personnage");
const choixDifficulte = document.querySelector("#difficulte");
const choixTheme = document.querySelector("#theme");
const boutonPouvoir = document.querySelector("#pouvoir");
const infoPouvoir = document.querySelector("#info-pouvoir");
const boutonRejouer = document.querySelector("#rejouer");

const combinaisonsGagnantes = [
 [0, 1, 2], [3, 4, 5], [6, 7, 8],
 [0, 3, 6], [1, 4, 7], [2, 5, 8],
 [0, 4, 8], [2, 4, 6]
];

let grille = Array(9).fill("");
let joueur;
let robot;
let partieTerminee = false;
let tourJoueur = true;

let modePouvoir = false;
let pouvoirJoueurUtilise = false;
let pouvoirRobotUtilise = false;

let caseBloqueeJoueur = -1;
let caseBloqueeRobot = -1;

let scores = { joueur: 0, robot: 0, nul: 0 };

function changerTheme() {
 document.body.className = "theme-" + choixTheme.value;
}

function trouverCombinaison(symbole) {
 return combinaisonsGagnantes.find(combinaison =>
 combinaison.every(index => grille[index] === symbole)
 ) || null;
}

function caseDisponible(index) {
 return (
 grille[index] === "" &&
 index !== caseBloqueeJoueur &&
 index !== caseBloqueeRobot
 );
}

function choisirAuHasard() {
 const libres = grille
 .map((valeur, index) => caseDisponible(index) ? index : -1)
 .filter(index => index !== -1);

 if (libres.length === 0) return -1;

 return libres[Math.floor(Math.random() * libres.length)];
}

function afficherGrille() {
 cases.forEach((bouton, index) => {
 bouton.classList.remove("bloquee", "gagnante");

 const bloquee =
 index === caseBloqueeJoueur ||
 index === caseBloqueeRobot ||
 grille[index] === "#";

 bouton.textContent = bloquee ? "🔒" : grille[index];
 bouton.classList.toggle("bloquee", bloquee);

 bouton.disabled =
 partieTerminee ||
 !tourJoueur ||
 grille[index] !== "" ||
 bloquee;

 bouton.setAttribute(
 "aria-label",
 `Case ${index + 1}${bloquee ? " bloquée" :
 grille[index] ? " : " + grille[index] : " libre"}`
 );
 });

 if (partieTerminee) {
 const combinaison =
 trouverCombinaison(joueur) ||
 trouverCombinaison(robot);

 if (combinaison) {
 combinaison.forEach(index =>
 cases[index].classList.add("gagnante")
 );
 }
 }
}

function verifierFin() {
 if (trouverCombinaison(joueur)) {
 terminerPartie("joueur");
 return true;
 }

 if (trouverCombinaison(robot)) {
 terminerPartie("robot");
 return true;
 }

 if (grille.every((valeur, index) =>
 valeur !== "" ||
 index === caseBloqueeJoueur ||
 index === caseBloqueeRobot
 )) {
 terminerPartie("nul");
 return true;
 }

 return false;
}

function terminerPartie(resultat) {
 partieTerminee = true;
 tourJoueur = false;
 modePouvoir = false;

 caseBloqueeJoueur = -1;
 caseBloqueeRobot = -1;

 if (resultat === "joueur") {
 message.textContent = "🎉 Bravo, tu as gagné !";
 scores.joueur++;
 } else if (resultat === "robot") {
 message.textContent = "L'adversaire a gagné !";
 scores.robot++;
 } else {
 message.textContent = "🤝 Match nul !";
 scores.nul++;
 }

 document.querySelector("#score-joueur").textContent = scores.joueur;
 document.querySelector("#score-robot").textContent = scores.robot;
 document.querySelector("#score-nul").textContent = scores.nul;

 boutonPouvoir.disabled = true;
 afficherGrille();
}

// Coups stratégiques
function trouverCoupStrategique(symbole) {
 for (let index = 0; index < grille.length; index++) {
 if (!caseDisponible(index)) continue;

 grille[index] = symbole;
 const gagne = trouverCombinaison(symbole) !== null;
 grille[index] = "";

 if (gagne) return index;
 }

 return -1;
}

function coupMoyen() {
 const attaque = trouverCoupStrategique(robot);
 if (attaque !== -1) return attaque;

 const defense = trouverCoupStrategique(joueur);
 if (defense !== -1) return defense;

 return choisirAuHasard();
}

function minimax(symboleQuiJoue) {
 if (trouverCombinaison(robot)) return 10;
 if (trouverCombinaison(joueur)) return -10;

 if (!grille.includes("")) return 0;

 const maximiser = symboleQuiJoue === robot;
 let meilleurScore = maximiser ? -Infinity : Infinity;

 for (let index = 0; index < grille.length; index++) {
 if (!caseDisponible(index)) continue;

 grille[index] = symboleQuiJoue;

 const prochainSymbole =
 symboleQuiJoue === robot ? joueur : robot;

 const score = minimax(prochainSymbole);
 grille[index] = "";

 meilleurScore = maximiser
 ? Math.max(meilleurScore, score)
 : Math.min(meilleurScore, score);
 }

 return meilleurScore;
}

function coupDifficile() {
 let meilleurCoup = -1;
 let meilleurScore = -Infinity;

 for (let index = 0; index < grille.length; index++) {
 if (!caseDisponible(index)) continue;

 grille[index] = robot;
 const score = minimax(joueur);
 grille[index] = "";

 if (score > meilleurScore) {
 meilleurScore = score;
 meilleurCoup = index;
 }
 }

 return meilleurCoup;
}

function activerPouvoir() {
 if (
 partieTerminee ||
 pouvoirJoueurUtilise ||
 !tourJoueur
 ) return;

 modePouvoir = !modePouvoir;

 message.textContent = modePouvoir
 ? "Choisis une case libre à bloquer."
 : "À toi de jouer !";

 infoPouvoir.textContent = modePouvoir
 ? "Le pouvoir ne consomme pas ton tour."
 : "Tu peux encore utiliser ton pouvoir.";
}

function pouvoirSpecialRobot() {
 if (pouvoirRobotUtilise) return false;

 const index = choisirAuHasard();
 if (index === -1) return false;

 caseBloqueeRobot = index;
 pouvoirRobotUtilise = true;

 message.textContent = "L'adversaire bloque une case !";
 afficherGrille();

 return true;
}

function jouer(index) {
 if (
 partieTerminee ||
 !tourJoueur ||
 !caseDisponible(index)
 ) return;

 if (modePouvoir) {
 caseBloqueeJoueur = index;
 modePouvoir = false;
 pouvoirJoueurUtilise = true;

 boutonPouvoir.disabled = true;
 infoPouvoir.textContent = "Pouvoir utilisé !";
 message.textContent = "Case bloquée ! À toi de jouer.";

 afficherGrille();
 return;
 }

 grille[index] = joueur;
 caseBloqueeRobot = -1;

 afficherGrille();
 if (verifierFin()) return;

 tourJoueur = false;
 message.textContent = "À l'adversaire de jouer...";
 afficherGrille();

 setTimeout(tourOrdinateur, 350);
}

function tourOrdinateur() {
 if (partieTerminee) return;

 if (!pouvoirRobotUtilise && Math.random() < 0.30) {
 pouvoirSpecialRobot();
 }

 let coup;

 if (choixDifficulte.value === "facile") {
 coup = choisirAuHasard();
 } else if (choixDifficulte.value === "moyen") {
 coup = coupMoyen();
 } else {
 coup = coupDifficile();
 }

 if (coup === -1) {
 caseBloqueeJoueur = -1;
 caseBloqueeRobot = -1;

 if (!verifierFin()) terminerPartie("nul");
 return;
 }

 grille[coup] = robot;
 caseBloqueeJoueur = -1;

 afficherGrille();
 if (verifierFin()) return;

 tourJoueur = true;
 message.textContent = "À toi de jouer !";
 afficherGrille();
}

function recommencer() {
 grille = Array(9).fill("");

 joueur = choixPersonnage.value;
 robot = joueur === "X" ? "O" : "X";

 partieTerminee = false;
 modePouvoir = false;
 pouvoirJoueurUtilise = false;
 pouvoirRobotUtilise = false;

 caseBloqueeJoueur = -1;
 caseBloqueeRobot = -1;

 scores = { joueur: 0, robot: 0, nul: 0 };

 document.querySelector("#score-joueur").textContent = 0;
 document.querySelector("#score-robot").textContent = 0;
 document.querySelector("#score-nul").textContent = 0;

 boutonPouvoir.disabled = false;
 infoPouvoir.textContent = "Utilise ton pouvoir une fois par partie.";

 changerTheme();

 tourJoueur = joueur === "X";
 afficherGrille();

 if (tourJoueur) {
 message.textContent = "À toi de jouer !";
 } else {
 message.textContent = "L'adversaire commence !";
 setTimeout(tourOrdinateur, 350);
 }
}

cases.forEach((bouton, index) => {
 bouton.addEventListener("click", () => jouer(index));
});

boutonPouvoir.addEventListener("click", activerPouvoir);
boutonRejouer.addEventListener("click", recommencer);

choixPersonnage.addEventListener("change", recommencer);
choixDifficulte.addEventListener("change", recommencer);
choixTheme.addEventListener("change", changerTheme);

recommencer();
