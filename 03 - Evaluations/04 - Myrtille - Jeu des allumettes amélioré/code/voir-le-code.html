let allumettes;
let joueuse = 1;
const noms = [];
const accueil = document.querySelector("#accueil");
const nom1 = document.querySelector("#nom1");
const nom2 = document.querySelector("#nom2");
const nombre = document.querySelector("#nombre");
const compteur = document.querySelector("#compteur");
const message = document.querySelector("#message");
const tas = document.querySelector("#tas");
const bouton = document.querySelector("#retirer");
const bouton2 = document.querySelector("#retirer2");
const bouton3 = document.querySelector("#retirer3");
const boutonr = document.querySelector("#recommencer");

document.querySelector("#commencer").addEventListener("click", function () {
  noms[1] = nom1.value || "Joueur 1";
  noms[2] = nom2.value || "Joueur 2";

  allumettes = Number(nombre.value);
  if (allumettes < 4) allumettes = 4;
  if (allumettes > 50) allumettes = 50;

  accueil.remove();
  message.textContent = "Au tour de " + noms[joueuse];
  message.className = "j" + joueuse;
  afficher();
});

bouton.addEventListener("click", function () {
  allumettes = allumettes - 1;
  test();
});
bouton2.addEventListener("click", function () {
  allumettes = allumettes - 2;
  test();
});
bouton3.addEventListener("click", function () {
  allumettes = allumettes - 3;
  test();
});
boutonr.addEventListener("click", function () {
  location.reload();
});

function afficher() {
  tas.innerHTML = "";
  for (let i = 0; i < allumettes; i++) {
    tas.innerHTML = tas.innerHTML + '<span class="allumette"></span>';
  }
  compteur.textContent = allumettes;
}

function test() {
  afficher();
  if (allumettes == 0) {
    message.textContent = noms[joueuse] + " a perdu !";
    bouton.disabled = true;
    bouton2.disabled = true;
    bouton3.disabled = true;
  } else {
    if (joueuse == 1) {
      joueuse = 2;
    } else {
      joueuse = 1;
    }
    message.textContent = "Au tour de " + noms[joueuse];
    message.className = "j" + joueuse;
    if (allumettes < 3) bouton3.disabled = true;
    if (allumettes < 2) bouton2.disabled = true;
  }
}
