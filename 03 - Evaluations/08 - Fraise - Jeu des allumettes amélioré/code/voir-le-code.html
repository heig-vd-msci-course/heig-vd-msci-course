let allumettes = 20;
let joueuse = 1;
let vies1 = 3;
let vies2 = 3;
let animationEnCours = false;

let prenom1 = "Joueuse 1";
let prenom2 = "Joueuse 2";


const accueil = document.querySelector("#accueil");
const choixPrenoms = document.querySelector("#choixPrenoms");
const interfaceJeu = document.querySelector("#interfaceJeu");

const jouer = document.querySelector("#jouer");
const commencer = document.querySelector("#commencer");


const affichageVies1 = document.querySelector("#vies1");
const affichageVies2 = document.querySelector("#vies2");
const compteur = document.querySelector("#compteur");
const message = document.querySelector("#message");
const tas = document.querySelector("#tas");

const retirer = document.querySelector("#retirer");
const retirer2 = document.querySelector("#retirer2");
const retirer3 = document.querySelector("#retirer3");

const rejouer = document.querySelector("#rejouer");
const nouvellePartie = document.querySelector("#nouvellePartie");
const trophee = document.querySelector("#trophee");
const confettis = document.querySelector("#confettis");


function nomJoueuse(numero) {
 return numero === 1 ? prenom1 : prenom2;
}


function afficherVies() {
 affichageVies1.textContent =
 prenom1 + " : " + "❤️".repeat(vies1);

 affichageVies2.textContent =
 prenom2 + " : " + "❤️".repeat(vies2);
}


function animerCoeur(numero) {
 const affichage = document.querySelector("#vies" + numero);
 const coeur = document.createElement("span");

 coeur.textContent = "💔";
 coeur.classList.add("coeur-brise");
 affichage.appendChild(coeur);

 setTimeout(function() {
 coeur.remove();
 }, 600);
}


function dessinerAllumettes() {
 tas.innerHTML = "";

 for (let i = 0; i < allumettes; i++) {
 tas.innerHTML += '<span class="allumette"></span>';
 }
}


function verifierBoutons() {
 const fin = vies1 <= 0 || vies2 <= 0;

 retirer.disabled =
 animationEnCours || fin || allumettes < 1;

 retirer2.disabled =
 animationEnCours || fin || allumettes < 2;

 retirer3.disabled =
 animationEnCours || fin || allumettes < 3;
}


function animerAllumettes(nombre) {
 const elements = tas.querySelectorAll(".allumette");

 for (let i = 0; i < nombre; i++) {
 const index = elements.length - 1 - i;

 if (elements[index]) {
 elements[index].classList.add("disparait");
 }
 }
}


function lancerConfettis() {
 confettis.innerHTML = "";

 const couleurs = [
 "#FF69B4",
 "#FFB6C1",
 "#FF0000",
 "#FFFFFF"
 ];

 for (let i = 0; i < 100; i++) {
 const confetti = document.createElement("div");

 confetti.classList.add("confetti");

 confetti.style.left = Math.random() * 100 + "%";

 confetti.style.backgroundColor =
 couleurs[Math.floor(Math.random() * couleurs.length)];

 confetti.style.animationDelay =
 Math.random() * 2 + "s";

 confettis.appendChild(confetti);
 }

 setTimeout(function() {
 confettis.innerHTML = "";
 }, 5000);
}


function musiqueVictoire() {
 const AudioContextClass =
 window.AudioContext || window.webkitAudioContext;

 if (!AudioContextClass) {
 return;
 }

 const audio = new AudioContextClass();
 const notes = [523, 659, 784, 1046];

 notes.forEach(function(note, i) {
 const oscillateur = audio.createOscillator();
 const volume = audio.createGain();

 oscillateur.connect(volume);
 volume.connect(audio.destination);

 oscillateur.type = "sine";
 oscillateur.frequency.value = note;

 const debut = audio.currentTime + i * 0.2;

 volume.gain.setValueAtTime(0.0001, debut);
 volume.gain.exponentialRampToValueAtTime(
 0.15, debut + 0.02
 );
 volume.gain.exponentialRampToValueAtTime(
 0.0001, debut + 0.18
 );

 oscillateur.start(debut);
 oscillateur.stop(debut + 0.2);
 });

 audio.resume();
}


function retirerAllumettes(nombre) {
 if (
 animationEnCours ||
 vies1 <= 0 ||
 vies2 <= 0 ||
 nombre > allumettes
 ) {
 return;
 }

 animationEnCours = true;

 animerAllumettes(nombre);
 verifierBoutons();

 setTimeout(function() {
 allumettes -= nombre;

 compteur.textContent = allumettes;
 dessinerAllumettes();

 if (allumettes === 0) {
 const perdante = joueuse;

 if (perdante === 1) {
 vies1--;
 } else {
 vies2--;
 }

 afficherVies();
 animerCoeur(perdante);

 if (vies1 === 0 || vies2 === 0) {
 const gagnante = perdante === 1 ? 2 : 1;

 message.textContent =
 "🏆 Bravo " + nomJoueuse(gagnante) +
 " ! Tu as gagné ! 🍓";

 trophee.style.display = "block";
 trophee.classList.remove("trophee-anime");

 void trophee.offsetWidth;

 trophee.classList.add("trophee-anime");

 lancerConfettis();
 musiqueVictoire();

 rejouer.style.display = "none";
 nouvellePartie.style.display = "block";

 } else {
 message.textContent =
 nomJoueuse(perdante) +
 " a perdu une manche ! Clique sur Rejouer.";
 }

 } else {
 joueuse = joueuse === 1 ? 2 : 1;

 message.textContent =
 "Au tour de " + nomJoueuse(joueuse);
 }

 animationEnCours = false;
 verifierBoutons();

 }, 400);
}

});

 confettis.innerHTML = "";

 nouvellePartie.style.display = "none";
 rejouer.style.display = "inline-block";

 verifierBoutons();
});
