let allumettes = 20;
let joueuse = 1;
let noms = { 1: "joueuse 1", 2: "joueuse 2" };
let blocsBleus = [];
const choixAllumettes = document.querySelector("#choixAllumettes");
const boutonCommencer = document.querySelector("#commencer");
const compteur = document.querySelector("#compteur");
const message = document.querySelector("#message");
const bouton = document.querySelector("#retirer");
const bouton2 = document.querySelector("#retirer2");
const bouton3 = document.querySelector("#retirer3");
const boutonRejouer = document.querySelector("#rejouer");
const champ1 = document.querySelector("#nom1");
const champ2 = document.querySelector("#nom2");
const boutonValider = document.querySelector("#valider");
const tas = document.querySelector("#tas");
function genererBlocsBleus() {
blocsBleus = [];

for (let i = 0; i < allumettes; i++) {
if (Math.random() < 0.15) {
blocsBleus.push(i);
}
}
if (blocsBleus.length > 4) {
const choisi =
blocsBleus[Math.floor(Math.random() * blocsBleus.length)];
blocsBleus = [choisi];
}
}


function afficherBlocs() {
tas.innerHTML = "";
for (let i = 0; i < allumettes; i++) {
const bloc = document.createElement("span");
bloc.className = "lucky-block";
if (blocsBleus.includes(i)) {
bloc.classList.add("bleu");
bloc.title = "Lucky Block spécial";
}
tas.appendChild(bloc);
}
compteur.textContent = allumettes;
verifierBoutons();
}


function verifierBoutons() {
bouton.disabled = allumettes < 1;
bouton2.disabled = allumettes < 2;
bouton3.disabled = allumettes < 3;
}

function changerJoueuse() {
joueuse = joueuse === 1 ? 2 : 1;
}

function effetBlocBleu() {
const variation = Math.floor(Math.random() * 9) - 4;
if (variation > 0) {
allumettes += variation;
message.textContent =
"Lucky Block spécial ! " + variation + " blocks ajoutés !";
} else if (variation < 0) {
allumettes = Math.max(0, allumettes + variation);
message.textContent =
"Lucky Block spécial ! " + Math.abs(variation) +
" blocks retirés !";
} else {
message.textContent =
"Lucky Block spécial ! Tu n'as rien eu :(";
}
}

function retirerBlocs(nombre) {
if (allumettes < nombre || allumettes <= 0) return;
const debut = allumettes - nombre;
const bleuTrouve = blocsBleus.some(position => position >= debut);
allumettes -= nombre;
blocsBleus = blocsBleus.filter(position => position < allumettes);
if (bleuTrouve) {
effetBlocBleu();
}
if (allumettes <= 0) {
allumettes = 0;
afficherBlocs();
message.textContent = noms[joueuse] + " a perdu !";
return;
}
changerJoueuse();
if (!bleuTrouve) {
message.textContent = "Au tour de " + noms[joueuse];
}
afficherBlocs();
}
bouton.addEventListener("click", () => retirerBlocs(1));
bouton2.addEventListener("click", () => retirerBlocs(2));
bouton3.addEventListener("click", () => retirerBlocs(3));
boutonValider.addEventListener("click", function () {
if (champ1.value.trim() !== "") {
noms[1] = champ1.value.trim();
}
if (champ2.value.trim() !== "") {
noms[2] = champ2.value.trim();
}
if (allumettes > 0) {
message.textContent = "Au tour de " + noms[joueuse];
}
});

function nouvellePartie(nombre) {
allumettes = nombre;
joueuse = 1;
genererBlocsBleus();
afficherBlocs();
message.textContent = "Au tour de " + noms[1];
}

nouvellePartie(20);

boutonRejouer.addEventListener("click", function () {
nouvellePartie(20);
});
boutonCommencer.addEventListener("click", function () {
const nombreChoisi = Number(choixAllumettes.value);
if (
choixAllumettes.value === "" ||
nombreChoisi < 1 ||
nombreChoisi > 100 ||
!Number.isInteger(nombreChoisi)
) {
message.textContent = "Choisis un nombre entre 1 et 100 !";
return;
}
nouvellePartie(nombreChoisi);
});
