function afficherListe(meteo, destination) {
    let affaires;

    if (meteo === "beau") {
        affaires = [
            ["Lunettes de soleil", "🕶️"],
            ["T-shirt", "👕"],
            ["Short", "🩳"],
            ["Crème solaire", "🧴"],
            ["Bouteille d'eau", "💧"],
            ["Chaussures confortables", "👟"],
            ["Casquette", "🧢"]
        ];
    } else {
        affaires = [
            ["Parapluie", "☂️"],
            ["Veste imperméable", "🧥"],
            ["Pantalon", "👖"],
            ["Pull chaud", "🧶"],
            ["Chaussures fermées", "👟"],
            ["Écharpe", "🧣"],
            ["Chaussettes", "🧦"]
        ];
    }

    const noms = {
        paris: "Paris",
        naples: "Naples",
        bruxelles: "Bruxelles"
    };

    const suffixes = {
        paris: "",
        naples: "-naples",
        bruxelles: "-bruxelles"
    };

    const suffixe = suffixes[destination];
    const message = document.getElementById("message" + suffixe);
    const liste = document.getElementById("liste" + suffixe);

    if (meteo === "beau") {
        message.textContent =
            "☀️ Beau temps à " + noms[destination] +
            " ! Voici les affaires à emporter.";
    } else {
        message.textContent =
            "🌧️ Temps pluvieux à " + noms[destination] +
            " ! Voici les affaires à emporter.";
    }

    liste.innerHTML = "";

    affaires.forEach(function(item) {
        liste.innerHTML += `
            <div class="affaire">
                <h2>${item[1]}</h2>
                <h3>${item[0]}</h3>
            </div>
        `;
    });
}
